<script>
    function calcularLucro() {
        const bruto = parseFloat(document.getElementById('ganhoBruto').value) || 0;
        const km = parseFloat(document.getElementById('kmRodados').value) || 0;
        const combustivel = parseFloat(document.getElementById('gastoCombustivel').value) || 0;
        const outras = parseFloat(document.getElementById('outrasDespesas').value) || 0;

        // Estimativa de reserva para manutenção e depreciação veicular (R$ 0.20 por km rodado)
        const depreciacaoManutencao = km * 0.20;

        const totalDespesas = combustivel + outras + depreciacaoManutencao;
        const lucroLiquido = bruto - totalDespesas;
        const custoPorKm = km > 0 ? (totalDespesas / km) : 0;

        document.getElementById('resGastos').innerText = 'R$ ' + totalDespesas.toFixed(2).replace('.', ',');
        document.getElementById('resCustoKm').innerText = 'R$ ' + custoPorKm.toFixed(2).replace('.', ',');
        document.getElementById('resLucro').innerText = 'R$ ' + lucroLiquido.toFixed(2).replace('.', ',');

        document.getElementById('painelResultado').style.display = 'block';
    }
</script>
